\newcommand{\journalitem}[1]{%
	\refstepcounter{journal}%
	\edef\@currentlabel{J\thejournal}%
	\label{#1}%
	\item[\textbf{J\thejournal}]%
}

\newcommand{\preprintitem}[1]{%
	\refstepcounter{preprint}%
	\edef\@currentlabel{P\thepreprint}%
	\label{#1}%
	\item[\textbf{P\thepreprint}]%
}

\newcommand{\conferenceitem}[1]{%
	\refstepcounter{conference}%
	\edef\@currentlabel{C\theconference}%
	\label{#1}%
	\item[\textbf{C\theconference}]%
}

\newcommand{\subconfitem}[1]{%
	\refstepcounter{submittedconference}%
	\edef\@currentlabel{S\thesubmittedconference}%
	\label{#1}%
	\item[\textbf{S\thesubmittedconference}]%
}

\newcommand{\pubref}[1]{%
	\textsuperscript{[\hyperref[#1]{\ref*{#1}}]}%
}
